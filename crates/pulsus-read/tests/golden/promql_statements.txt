== max by (status) (http_requests_total{status="500"}) ==
params start_ms=1782907200000 end_ms=1782928800000 step_ms=60000
selectors 1
-- selector[0] name=http_requests_total range_ms=- window=(1782906900000, 1782928800000]
-- grouped op=Max gids=[0, 1, 0]
-- statement[004] offset=00002211 len=002682
WITH 1782907200000 AS grid_start, 60000 AS grid_step, 361 AS grid_n, 300000 AS lookback,
     [toUInt128('101'), toUInt128('205'), toUInt128('990')] AS fps,
     CAST([0, 1, 0], 'Array(UInt32)') AS gids
SELECT gid, min(gi) AS gi_start, max(gi) AS gi_end, any(agg) AS agg, any(flags) AS flags
FROM (
  SELECT gid, gi, agg, flags,
    sum(is_new) OVER (PARTITION BY gid ORDER BY gi ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS run
  FROM (
    SELECT gid, gi, agg, flags,
      toUInt8(gi != lagInFrame(gi) OVER w + 1
              OR reinterpretAsUInt64(agg) != reinterpretAsUInt64(lagInFrame(agg) OVER w)
              OR flags != lagInFrame(flags) OVER w) AS is_new
    FROM (
      SELECT gid, gi,
        if(countIf(NOT is_hist AND NOT isNaN(v)) = 0, argMaxIf(v, transform(fingerprint, fps, arrayEnumerate(fps), toUInt32(0)), NOT is_hist),
           maxIf(v, NOT is_hist AND NOT isNaN(v))) AS agg,
        toUInt8(if(countIf(NOT is_hist) > 0, 1, 0) + if(countIf(is_hist) > 0, 2, 0)) AS flags
      FROM (
        SELECT gid, fingerprint, v, is_hist,
          arrayJoin(range(
            toUInt32(least(toInt64(grid_n),
              if(ts <= grid_start, 0, intDiv(ts - grid_start + grid_step - 1, grid_step)))),
            toUInt32(least(toInt64(grid_n),
              if(cover_end <= grid_start, 0, intDiv(cover_end - grid_start + grid_step - 1, grid_step))))
          )) AS gi
        FROM (
          SELECT transform(fingerprint, fps, gids, CAST(0, 'UInt32')) AS gid, fingerprint,
            ts, v, is_hist, stale,
            least(leadInFrame(ts, 1, toInt64(1782928800000) + lookback + 1) OVER (
                    PARTITION BY fingerprint ORDER BY ts, is_hist
                    ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING),
                  ts + lookback) AS cover_end
          FROM (
            SELECT fingerprint, unix_milli AS ts, value AS v, CAST(0, 'UInt8') AS is_hist,
                   reinterpretAsUInt64(value) = 9218868437227405314 AS stale
            FROM metric_samples
            WHERE unix_milli > 1782906900000 AND unix_milli <= 1782928800000 AND fingerprint IN fps
            UNION ALL
            SELECT fingerprint, unix_milli AS ts, CAST(0, 'Float64') AS v, CAST(1, 'UInt8') AS is_hist,
                   reinterpretAsUInt64(sum) = 9218868437227405314 AS stale
            FROM metric_hist_samples
            WHERE unix_milli > 1782906900000 AND unix_milli <= 1782928800000 AND fingerprint IN fps
          )
        )
        WHERE NOT stale
      )
      GROUP BY gi, gid
    )
    WINDOW w AS (PARTITION BY gid ORDER BY gi ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
  )
)
GROUP BY gid, run
ORDER BY gid, gi_start

== sum by (status) (rate(http_requests_total{status="500"}[5m])) ==
params start_ms=1782907200000 end_ms=1782928800000 step_ms=60000
selectors 1
-- selector[0] name=http_requests_total range_ms=300000 window=(1782906600000, 1782928800000]
-- pushed op=Sum func=Rate gids=[0, 1, 0]
-- statement[026] offset=00022107 len=004643
WITH 1782907200000 AS grid_start, 60000 AS grid_step, 361 AS grid_n, 300000 AS range_ms,
     [toUInt128('101'), toUInt128('205'), toUInt128('990')] AS fps,
     CAST([0, 1, 0], 'Array(UInt32)') AS gids
SELECT gid, gi, (arrayFold((acc, x) -> (acc.1 + x, if(isInfinite(acc.1 + x), 0., if(abs(acc.1) >= abs(x), acc.2 + ((acc.1 - (acc.1 + x)) + x), acc.2 + ((x - (acc.1 + x)) + acc.1)))), vs, CAST((0., 0.), 'Tuple(Float64, Float64)')) AS k).1 + k.2 AS agg
FROM (
  SELECT gid, gi, arrayMap(p -> p.2, arraySort(groupArray((transform(fingerprint, fps, arrayEnumerate(fps), toUInt32(0)), v)))) AS vs

    SELECT transform(fingerprint, fps, gids, CAST(0, 'UInt32')) AS gid, fingerprint, gi,
        i_l - i_f + 1 AS n,
        arrayFold((acc, x) -> acc + x.2, arrayFilter(x -> x.1 > i_f, r_l),
                  if(y_f < 0., (y_l - y_f) + 0., y_l - y_f)) AS result,
        toFloat64(t_f - (grid_start + gi * grid_step - range_ms)) / 1000. AS d_start_raw,
        toFloat64((grid_start + gi * grid_step) - t_l) / 1000. AS d_end_raw,
        toFloat64(t_l - t_f) / 1000. AS sampled,
        sampled / toFloat64(n - 1) AS avg_dur,
        avg_dur * 1.1 AS threshold,
        if(d_start_raw >= threshold, avg_dur / 2., d_start_raw) AS d_start_1,
        if(result > 0. AND y_f >= 0., sampled * (y_f / result), d_start_1) AS d_zero,
        if(d_zero < d_start_1, d_zero, d_start_1) AS d_start,
        if(d_end_raw >= threshold, avg_dur / 2., d_end_raw) AS d_end,
        ((sampled + d_start + d_end) / sampled) / (toFloat64(range_ms) / 1000.) AS factor,
        result * factor AS v
    FROM (
      SELECT fingerprint, gi, i_f, t_f, y_f, i AS i_l, ts AS t_l, y AS y_l, ts_prev AS t_p, y_prev AS y_p, resets AS r_l
      FROM (
        SELECT fingerprint, gi, role, i, ts, y, ts_prev, y_prev, resets,
          lagInFrame(i) OVER wr AS i_f, lagInFrame(ts) OVER wr AS t_f, lagInFrame(y) OVER wr AS y_f
        FROM (
          SELECT fingerprint, i, ts, y, ts_prev, y_prev, resets, role,
            arrayJoin(range(
              toUInt32(least(toInt64(grid_n), if(a <= grid_start, 0, intDiv(a - grid_start + grid_step - 1, grid_step)))),
              toUInt32(least(toInt64(grid_n), if(b <= grid_start, 0, intDiv(b - grid_start + grid_step - 1, grid_step)))))) AS gi
          FROM (
            SELECT fingerprint, i, ts, y, ts_prev, y_prev, resets, role,
              if(role = 0, if(i = 1, ts, greatest(ts, ts_prev + range_ms)), ts) AS a,
              if(role = 0, ts + range_ms, least(ts_next, ts + range_ms)) AS b
            FROM (
              SELECT fingerprint, ts, y, i, ts_prev, y_prev, ts_next,
                groupArrayIf((i, y_prev), i > 1 AND y < y_prev) OVER w AS resets
              FROM (
                SELECT fingerprint, ts, y, yb,
                  row_number() OVER w AS i,
                  lagInFrame(ts, 1, toInt64(0)) OVER w AS ts_prev,
                  lagInFrame(y, 1, 0.) OVER w AS y_prev,
                  leadInFrame(ts, 1, toInt64(9223372036854775807)) OVER wf AS ts_next
                FROM (
                  SELECT fingerprint, unix_milli AS ts, reinterpretAsUInt64(value) AS yb, value AS y,
                    ts = lagInFrame(unix_milli, 1, toInt64(-1)) OVER w0
                      AND yb = lagInFrame(reinterpretAsUInt64(value), 1, toUInt64(0)) OVER w0 AS dup
                  FROM metric_samples
                  WHERE unix_milli > 1782906900000 AND unix_milli <= 1782928800000 AND fingerprint IN fps
                    AND reinterpretAsUInt64(value) != 9218868437227405314
                  WINDOW w0 AS (PARTITION BY fingerprint ORDER BY unix_milli, reinterpretAsUInt64(value)
                                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
                )
                WHERE NOT dup
                WINDOW w AS (PARTITION BY fingerprint ORDER BY ts, yb ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW),
                       wf AS (PARTITION BY fingerprint ORDER BY ts, yb ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING)
              )
              WINDOW w AS (PARTITION BY fingerprint ORDER BY ts, yb ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
            ) ARRAY JOIN [0, 1] AS role
          )
        )
        WINDOW wr AS (PARTITION BY fingerprint ORDER BY gi, role ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
      )
      WHERE role = 1
    )
    WHERE i_l > i_f
  )
  GROUP BY gid, gi
)
UNION ALL
SELECT toUInt32(4294967295) AS gid, toUInt32(0) AS gi,
  toFloat64((SELECT count() FROM metric_hist_samples
             WHERE unix_milli > 1782906900000 AND unix_milli <= 1782928800000 AND fingerprint IN fps)) AS agg
ORDER BY gid, gi
